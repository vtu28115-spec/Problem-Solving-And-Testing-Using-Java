class Solution {
    public boolean halvesAreAlike(String s) {
        int mid = s.length() / 2, count = 0;

        for (int i = 0; i < mid; i++)
            if ("aeiouAEIOU".indexOf(s.charAt(i)) >= 0) count++;

        for (int i = mid; i < s.length(); i++)
            if ("aeiouAEIOU".indexOf(s.charAt(i)) >= 0) count--;

        return count == 0;
    }
}
